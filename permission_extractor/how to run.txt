cd permission_extractor
pip install -r requirements.txt
APK_ROOT=/path/to/android-apks ./run_mldp.sh

artifacts/metrics/evaluation_results.json      val + dev_test scores

artifacts/export/mldp_pruned_permission/       ONNX bundle for VigiDroid later
